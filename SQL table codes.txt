1 - creates the database

CREATE DATABASE rihla;


2 - creates the users table 

CREATE TABLE users (
  id INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  email VARCHAR(150) NOT NULL UNIQUE,
  password VARCHAR(255) NOT NULL,
  is_student TINYINT(1) DEFAULT 0,
  student_id VARCHAR(20) DEFAULT NULL
);


3 - creates the tickets table

CREATE TABLE user_tickets (
  id INT AUTO_INCREMENT PRIMARY KEY,
  user_id INT NOT NULL,
  ticket_name VARCHAR(100) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  purchased_at DATETIME NOT NULL,
  expires_at DATETIME NOT NULL,
  FOREIGN KEY (user_id) REFERENCES users(id)
);


4 - add a balance column in the users table

ALTER TABLE users
ADD balance DECIMAL(10,2) NOT NULL DEFAULT 0.00;


5 - create the wallet transaction table


CREATE TABLE wallet_transactions (
  id INT AUTO_INCREMENT PRIMARY KEY,
  user_id INT NOT NULL,
  type VARCHAR(30) NOT NULL,
  amount DECIMAL(10,2) NOT NULL,
  description VARCHAR(150) NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id)
);

5 - make the student ID unique

ALTER TABLE users
ADD CONSTRAINT unique_student_id UNIQUE (student_id);

6 - create the notification table

CREATE TABLE notifications (

    id INT AUTO_INCREMENT PRIMARY KEY,

    title VARCHAR(255) NOT NULL,

    message TEXT NOT NULL,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    status ENUM('active', 'inactive') DEFAULT 'active'

);


7 - create the stations table

CREATE TABLE stations (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    line VARCHAR(50) NOT NULL,
    latitude DECIMAL(10,7) NOT NULL,
    longitude DECIMAL(10,7) NOT NULL
);

8 - create the saved_stations table

CREATE TABLE saved_stations (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    station_id INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (station_id) REFERENCES stations(id) ON DELETE CASCADE
);
